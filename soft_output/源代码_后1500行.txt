          }
        }
        .alignItems(HorizontalAlign.Start);
        // 帖子网格
        if (this.posts.length === 0 && this.loading) {
          Column() {
            LoadingProgress().width(28).height(28);
            Text('加载中…').fontSize(13).fontColor($r('app.color.text_secondary')).margin({ top: 10 });
          }
          .layoutWeight(1).width('100%').justifyContent(FlexAlign.Center);
        } else if (this.posts.length === 0) {
          Column() {
            Text(this.error ? ('加载失败：' + this.error) : 'TA 还没有发布过帖子')
              .fontSize(14).fontColor($r('app.color.text_secondary'));
          }
          .layoutWeight(1).width('100%').justifyContent(FlexAlign.Center);
        } else {
          Refresh({ refreshing: this.isRefreshing, builder: this.refreshBuilder }) {
            Grid() {
              ForEach(this.posts, (post: Post) => {
                GridItem() {
                  PostCard({ post: post })
                    .onClick(() => {
                      this.goDetail(post);
                    });
                }
              }, (post: Post) => post.id !== undefined ? post.id.toString() : '');
              if (this.finished) {
                GridItem() {
                  Text('没有更多了')
                    .fontSize(12).fontColor($r('app.color.text_secondary'))
                    .width('100%').textAlign(TextAlign.Center).padding(16);
                }
                .columnStart(0).columnEnd(1);
              }
            }
            .columnsTemplate('1fr 1fr')
            .columnsGap(12).rowsGap(12).padding(12)
            .onScrollIndex((_start: number, end: number) => {
              if (end >= this.posts.length - 1) {
                this.loadMore();
              }
            });
          }
          .onRefreshing(() => {
            this.isRefreshing = true;
            this.refresh().finally(() => {
              this.isRefreshing = false;
            });
          })
          .layoutWeight(1).width('100%');
        }
      }
    }
    .width('100%').padding({ top: this.statusBarHeight }).height('100%').backgroundColor($r('app.color.bg_primary'))
    .transition(buildPageTransition())
  }
}

// - 真机/同局域网直连：'http://<你电脑局域网IP>:3000'
// - 上架后：'https://api.youju.com'
export const BASE_URL = 'http://127.0.0.1:3000';
// token 由 utils/auth.ts 统一管理（PersistentStorage 持久化），request 直接从 AppStorage 读取。
export interface ApiResult<T> {
  code: number;
  data: T;
  message: string;
}
// 判断错误是否为网络连接类失败（后端未启动 / hdc rport 未建立 / 超时等）
function isNetworkConnectionError(message: string): boolean {
  const m: string = message.toLowerCase();
  return m.includes('connect') || m.includes('refused') || m.includes('econnrefused') ||
    m.includes('unreachable') || m.includes('timed out') || m.includes('network');
}
async function request<T>(method: http.RequestMethod, path: string, body?: Object): Promise<T> {
  const url = BASE_URL + path;
  const req = http.createHttp();
  const token: string = (AppStorage.get<string>('authToken') ?? '') as string;
  const header: Record<string, string> = { 'Content-Type': 'application/json' };
  if (token) {
    header['Authorization'] = 'Bearer ' + token;
  }
  try {
    const resp = await req.request(url, {
      method,
      header,
      extraData: body ? JSON.stringify(body) : undefined,
      expectDataType: http.HttpDataType.OBJECT,
      connectTimeout: 10000,
      readTimeout: 10000,
    });
    // 先尝试解析 body 中的业务错误（后端 400/409/404 等返回 {code, message}）
    // 让后端业务错误 message 透传到前端 catch 块（如"内容含敏感词"、"你已举报过该内容"）
    const result = resp.result as ApiResult<T>;
    if (result && typeof result.code === 'number') {
      if (result.code !== 0) {
        throw new Error(result.message || '请求失败');
      }
      return result.data;
    }
    // fallback：非 JSON 响应或无 code 字段
    if (resp.responseCode !== 200) {
      throw new Error('HTTP ' + resp.responseCode);
    }
    throw new Error('请求失败');
  } catch (e) {
    const err: Error = e as Error;
    const rawMsg: string = err.message ? err.message : '请求发生未知错误';
    // 连接失败（ECONNREFUSED 等）给出可操作提示，避免直接透传系统底层错误
    if (isNetworkConnectionError(rawMsg)) {
      throw new Error('无法连接后端服务，请确认后端已启动且 hdc rport 已建立（模拟器重启后需重跑 rport）');
    }
    throw new Error(rawMsg);
  } finally {
    req.destroy();
  }
}
interface ApiClient {
  get: <T>(path: string) => Promise<T>;
  post: <T>(path: string, body?: Object) => Promise<T>;
  put: <T>(path: string, body?: Object) => Promise<T>;
  del: <T>(path: string) => Promise<T>;
}
export const api: ApiClient = {
  get: <T>(path: string): Promise<T> => request<T>(http.RequestMethod.GET, path),
  post: <T>(path: string, body?: Object): Promise<T> => request<T>(http.RequestMethod.POST, path, body),
  put: <T>(path: string, body?: Object): Promise<T> => request<T>(http.RequestMethod.PUT, path, body),
  del: <T>(path: string): Promise<T> => request<T>(http.RequestMethod.DELETE, path),
};

// 腾讯云 COS 直传签名：后端签发预签名 PUT URL，前端直传二进制
export interface UploadSignature {
  url: string; // 预签名 PUT URL（前端直传目标）
  key: string; // 对象 Key
  cdnUrl: string; // 直传后公开可读时访问的 URL
  viewUrl: string; // GET 预签名可读链接（私有桶也能加载，展示与存储一律用这个）
  contentType: string; // 前端 PUT 必须带上的 Content-Type（需与签名一致）
  mode?: 'cos' | 'local'; // 上传模式：cos=直传腾讯云(外网URL)；local=直传后端 /v1/upload/local
}
// 解析「前端 PUT 直传目标 URL」：
// - cos 模式：直传后端下发的外部预签名 URL（如 myqcloud.com）。
// - local 模式：上传必须走 rport(BASE_URL=127.0.0.1:3000)，与所有 API 调用一致；
//   而展示用的 viewUrl 仍用 LAN IP(backendPublicUrl)——因为模拟器 Image/WebView 不走 rport。
// 严禁在 local 模式直接用 sig.url（LAN IP），那会因模拟器 http 模块连不上而上传失败。
export function resolveUploadPutUrl(sig: UploadSignature): string {
  if (sig.mode === 'cos') {
    return sig.url;
  }
  if (sig.mode === 'local' || !sig.url.includes('myqcloud.com')) {
    const pathAndQuery: string = sig.url.replace(/^https?:\/\/[^/]+/, '');
    return BASE_URL + pathAndQuery;
  }
  return sig.url;
}
// 拼接 query string（跳过 undefined/null/空串）
function toQuery(fields: Array<[string, string | number | undefined]>): string {
  const parts: string[] = [];
  for (const f of fields) {
    const v = f[1];
    if (v !== undefined && v !== null && v !== '') {
      parts.push(f[0] + '=' + encodeURIComponent(String(v)));
    }
  }
  return parts.length > 0 ? '?' + parts.join('&') : '';
}

// 确保登录态：有登录态直接返回当前用户；无登录态则跳到 LoginPage 由用户显式登录，并返回 null。
// 不再静默用 dev 桩建号（原 MVP 行为），避免真实登录入口被绕过。
export async function ensureLogin(): Promise<User | null> {
  const existing: User | null = getCurrentUser();
  if (existing && getToken()) {
    return existing;
  }
  // 防递归：LoginPage 自身（如「其他方式登录」）调用时不再重复 pushUrl
  try {
    const state = router.getState();
    if (state && state.path !== 'pages/LoginPage') {
      router.pushUrl({ url: 'pages/LoginPage' });
    }
  } catch (e) {
    // router 未就绪等异常时直接尝试跳转，不阻断
    try {
      router.pushUrl({ url: 'pages/LoginPage' });
    } catch (_) { /* ignore */ }
  }
  return null;
}

// 响应式断点系统：根据窗口宽度返回适配参数
// 用于 Grid 列数、字体缩放、间距调整等
// 断点定义
const BREAKPOINT_SM = 600;   // vp: 手机竖屏
const BREAKPOINT_MD = 840;   // vp: 手机横屏 / 平板竖屏
const BREAKPOINT_LG = 1024;  // vp: 平板横屏 / 桌面
export enum Breakpoint {
  SM = 'sm',
  MD = 'md',
  LG = 'lg',
}
// 根据窗口宽度获取当前断点
export function getBreakpoint(width: number): Breakpoint {
  if (width < BREAKPOINT_SM) {
    return Breakpoint.SM;
  }
  if (width < BREAKPOINT_MD) {
    return Breakpoint.MD;
  }
  return Breakpoint.LG;
}
// 根据断点获取 Grid 列数
export function getGridColumns(bp: Breakpoint): number {
  switch (bp) {
    case Breakpoint.SM:
      return 2;
    case Breakpoint.MD:
      return 3;
    case Breakpoint.LG:
      return 4;
    default:
      return 2;
  }
}
// 根据断点获取 Grid 列模板字符串
export function getGridTemplate(bp: Breakpoint): string {
  const cols: number = getGridColumns(bp);
  return '1fr '.repeat(cols).trim();
}

// ========== ./entry/src/main/ets/utils/data-source.ets ==========
// 通用 LazyForEach 数据源，支持列表数据的懒加载渲染
// 替代 ForEach，在大量数据场景下显著提升性能
export class BasicDataSource<T> implements IDataSource {
  private listeners: DataChangeListener[] = [];
  private dataArray: T[] = [];
  totalCount(): number {
    return this.dataArray.length;
  }
  getData(index: number): T {
    return this.dataArray[index] as T;
  }
  registerDataChangeListener(listener: DataChangeListener): void {
    if (this.listeners.indexOf(listener) < 0) {
      this.listeners.push(listener);
    }
  }
  unregisterDataChangeListener(listener: DataChangeListener): void {
    const pos: number = this.listeners.indexOf(listener);
    if (pos >= 0) {
      this.listeners.splice(pos, 1);
    }
  }
  // 通知数据变更（用于刷新场景）
  notifyDataReload(): void {
    this.listeners.forEach((listener: DataChangeListener) => {
      listener.onDataReloaded();
    });
  }
  notifyDataAdd(index: number): void {
    this.listeners.forEach((listener: DataChangeListener) => {
      listener.onDataAdd(index);
    });
  }
  notifyDataChange(index: number): void {
    this.listeners.forEach((listener: DataChangeListener) => {
      listener.onDataChange(index);
    });
  }
  notifyDataDelete(index: number): void {
    this.listeners.forEach((listener: DataChangeListener) => {
      listener.onDataDelete(index);
    });
  }
  // 清空数据并通知
  clearData(): void {
    this.dataArray = [];
    this.notifyDataReload();
  }
  // 获取数据长度（用于空状态判断）
  getLength(): number {
    return this.dataArray.length;
  }
  // 设置数据
  setData(data: T[]): void {
    this.dataArray = data;
    this.notifyDataReload();
  }
  // 追加数据
  appendData(data: T[]): void {
    for (let i = 0; i < data.length; i++) {
      this.dataArray.push(data[i]);
      this.notifyDataAdd(this.dataArray.length - 1);
    }
  }
  // 替换全部数据
  replaceData(data: T[]): void {
    this.dataArray = data;
  }
  // 获取原始数据（用于外部访问）
  getRawData(): T[] {
    return this.dataArray;
  }
}

// ========== ./entry/src/main/ets/utils/format.ets ==========
export function formatCount(n?: number): string {
  if (!n) {
    return '0';
  }
  if (n >= 10000) {
    return (n / 10000).toFixed(1) + 'w';
  }
  return n.toString();
}
export function timeAgo(iso?: string): string {
  if (!iso) {
    return '';
  }
  const diff = Date.now() - new Date(iso).getTime();
  const m = Math.floor(diff / 60000);
  if (m < 1) {
    return '刚刚';
  }
  if (m < 60) {
    return m + '分钟前';
  }
  const h = Math.floor(m / 60);
  if (h < 24) {
    return h + '小时前';
  }
  const d = Math.floor(h / 24);
  if (d < 30) {
    return d + '天前';
  }
  return new Date(iso).toLocaleDateString();
}

// ========== ./entry/src/main/ets/utils/image.ets ==========
// 图片压缩：选图后长边压缩 + 质量 80% 转 jpeg（详见技术文档 5.3）。
// 同时返回 PixelMap（本地预览）与 bytes（上传 COS 用），一举两得。
import { image } from '@kit.ImageKit';
import { fileIo } from '@kit.CoreFileKit';
export interface CompressedImage {
  preview: image.PixelMap; // 本地预览用
  bytes: ArrayBuffer; // 上传二进制用（jpeg）
}
export async function compressImage(uri: string): Promise<CompressedImage> {
  // 相册返回的 photo:// URI 是沙箱受限 URI，必须经由文件描述符打开，
  // 直接用 image.createImageSource(uri) 在某些 SDK 会返回 undefined（导致 createPixelMap 报错）。
  const file = fileIo.openSync(uri, fileIo.OpenMode.READ_ONLY);
  let source: image.ImageSource;
  try {
    source = image.createImageSource(file.fd);
  } catch (e) {
    fileIo.closeSync(file);
    throw new Error((e as Error).message);
  }
  try {
    const pixelMap: image.PixelMap = await source.createPixelMap();
    const packer: image.ImagePacker = image.createImagePacker();
    try {
      const bytes: ArrayBuffer = await packer.packToData(pixelMap, {
        format: 'image/jpeg',
        quality: 80,
      });
      return { preview: pixelMap, bytes };
    } finally {
      await packer.release();
    }
  } finally {
    source.release();
    fileIo.closeSync(file);
  }
}
// 头像专用压缩：限制最长边 maxEdge（默认 512px），避免本地 base64 兜底时体积过大，质量 80%。
// 其余逻辑与 compressImage 一致（file descriptor 打开 photo:// URI）。
export async function compressAvatar(uri: string, maxEdge: number = 512): Promise<CompressedImage> {
  const file = fileIo.openSync(uri, fileIo.OpenMode.READ_ONLY);
  let source: image.ImageSource;
  try {
    source = image.createImageSource(file.fd);
  } catch (e) {
    fileIo.closeSync(file);
    throw new Error((e as Error).message);
  }
  try {
    const pixelMap: image.PixelMap = await source.createPixelMap();
    const info = pixelMap.getImageInfoSync();
    const longEdge: number = Math.max(info.size.width, info.size.height);
    if (longEdge > maxEdge) {
      const ratio: number = maxEdge / longEdge;
      await pixelMap.scale(ratio, ratio);
    }
    const packer: image.ImagePacker = image.createImagePacker();
    try {
      const bytes: ArrayBuffer = await packer.packToData(pixelMap, {
        format: 'image/jpeg',
        quality: 80,
      });
      return { preview: pixelMap, bytes };
    } finally {
      await packer.release();
    }
  } finally {
    source.release();
    fileIo.closeSync(file);
  }
}

// ========== ./entry/src/main/ets/utils/likeStore.ets ==========
// 全局点赞状态仓库（单例）
//
// 背景：点赞态此前只存在每个 PostCard / DetailPage 各自的 @State 里，首页卡片点赞后，
// 详情页、个人主页「我赞过」里的同一帖子副本完全不知道，导致跨页点赞数/已赞态不一致，
// 且「我赞过」列表在取消点赞后不会即时消失。
//
// 方案：用一个进程级单例保存 postId -> { liked, count }，所有展示帖子的组件在 aboutToAppear
// 时向仓库「注册并订阅」，点赞/取消时写入仓库并广播；订阅回调把最新值写回组件本地 @State，
// 触发重渲染。这是 ArkUI V1 严格模式下最稳的跨组件实时同步方式（不依赖 @Observed 深层观察）。
export class LikeInfo {
  liked: boolean = false;
  count: number = 0;
  constructor(liked: boolean, count: number) {
    this.liked = liked;
    this.count = count;
  }
}
type LikeListener = (postId: number) => void;
class LikeStoreImpl {
  private map: Map<number, LikeInfo> = new Map<number, LikeInfo>();
  private listeners: LikeListener[] = [];
  // 仅当仓库中尚无该帖子时灌入初始态（不覆盖本地已发生的点赞操作）
  ensure(postId: number, liked: boolean, count: number): void {
    if (!this.map.has(postId)) {
      this.map.set(postId, new LikeInfo(liked, count));
    }
  }
  get(postId: number): LikeInfo | undefined {
    return this.map.get(postId);
  }
  // 乐观更新（点赞/取消）。返回操作前的快照（LikeInfo 实例），便于请求失败时回滚。
  applyOptimistic(postId: number, willLike: boolean): LikeInfo {
    let info: LikeInfo | undefined = this.map.get(postId);
    if (!info) {
      info = new LikeInfo(false, 0);
      this.map.set(postId, info);
    }
    const prev: LikeInfo = new LikeInfo(info.liked, info.count);
    const delta: number = willLike ? 1 : (info.liked ? -1 : 0);
    info.liked = willLike;
    info.count = Math.max(0, info.count + delta);
    this.emit(postId);
    return prev;
  }
  // 请求失败回滚到操作前快照
  revert(postId: number, prev: LikeInfo): void {
    const info: LikeInfo | undefined = this.map.get(postId);
    if (info) {
      info.liked = prev.liked;
      info.count = prev.count;
      this.emit(postId);
    }
  }
  // 用服务器返回的绝对值覆盖（进入详情页 / 刷新列表后校准，避免与本地乐观值漂移）
  syncFromServer(postId: number, liked: boolean, count: number): void {
    let info: LikeInfo | undefined = this.map.get(postId);
    if (!info) {
      info = new LikeInfo(liked, count);
      this.map.set(postId, info);
    } else {
      info.liked = liked;
      info.count = count;
    }
    this.emit(postId);
  }
  // 订阅某帖子点赞态变化；返回取消订阅函数（组件 aboutToDisappear 时调用，防泄漏）
  subscribe(cb: LikeListener): () => void {
    this.listeners.push(cb);
    return (): void => {
      const idx: number = this.listeners.indexOf(cb);
      if (idx >= 0) {
        this.listeners.splice(idx, 1);
      }
    };
  }
  private emit(postId: number): void {
    for (const cb of this.listeners) {
      cb(postId);
    }
  }
}
export const likeStore: LikeStoreImpl = new LikeStoreImpl();

// ========== ./entry/src/main/ets/utils/responsive.ets ==========
// 响应式网格工具：根据屏幕宽度计算卡片列数，并监听窗口尺寸变化（折叠屏/平板/旋转）
// 用途：让首页/个人主页/搜索/圈子的卡片网格在不同屏幕尺寸下列数自适应，避免溢出或布局错乱
import { window } from '@kit.ArkUI';
import { common } from '@kit.AbilityKit';
// 根据屏幕物理宽度（px）计算网格列数
//   手机竖屏 (<600vp)        → 2 列
//   折叠屏展开/平板竖屏       → 3 列
//   平板横屏/桌面 (>840vp)    → 4 列
function computeGridColumns(widthPx: number): number {
  const wvp: number = px2vp(widthPx);
  if (wvp >= 840) {
    return 4;
  }
  if (wvp >= 600) {
    return 3;
  }
  return 2;
}
// 生成 Grid 的 columnsTemplate 字符串，例如 3 列 → '1fr 1fr 1fr'
export function gridColumnTemplate(cols: number): string {
  let tpl: string = '';
  for (let i: number = 0; i < cols; i++) {
    if (i > 0) {
      tpl += ' ';
    }
    tpl += '1fr';
  }
  return tpl;
}
// 注册窗口尺寸监听：尺寸变化时通过 onChange 回传最新列数；返回注销函数，须在 aboutToDisappear 调用
// ArkTS V1 约束：回调用具名函数类型 (size: window.Size) => void，禁止内联对象字面量
export function observeGridColumns(ctx: common.Context, onChange: (cols: number) => void): () => void {
  let handler: ((size: window.Size) => void) | undefined = undefined;
  window.getLastWindow(ctx)
    .then((win: window.Window) => {
      const rect = win.getWindowProperties().windowRect;
      onChange(computeGridColumns(rect.width));
      handler = (size: window.Size): void => {
        onChange(computeGridColumns(size.width));
      };
      win.on('windowSizeChange', handler);
    })
    .catch(() => {
      onChange(2);
    });
  return (): void => {
    if (handler !== undefined) {
      window.getLastWindow(ctx)
        .then((win: window.Window) => {
          win.off('windowSizeChange', handler);
        })
        .catch(() => { /* 忽略注销失败 */ });
    }
  };
}

// ========== ./entry/src/main/ets/utils/theme.ets ==========
// 主题工具：深色/浅色模式切换
// 策略（最保守版）：
// - setFollowSystem 不调 setColorMode，只更新 AppStorage + preferences
// - toggleDarkMode 调 setColorMode(DARK/LIGHT) 强制指定
// - onConfigurationUpdate 只更新缓存，不调 setColorMode
// - 消除所有 setColorMode 递归/异常路径
import { ConfigurationConstant, common } from '@kit.AbilityKit';
import { preferences } from '@kit.ArkData';
const PREF_NAME = 'youju_theme';
const KEY_FOLLOW_SYSTEM = 'followSystem';
const KEY_IS_DARK = 'isDarkMode';
let _appContext: common.ApplicationContext | null = null;
let _prefs: preferences.Preferences | null = null;
export function initThemeContext(ctx: common.ApplicationContext): void {
  _appContext = ctx;
  try {
    _prefs = preferences.getPreferencesSync(ctx, { name: PREF_NAME });
  } catch (e) {
  }
}
export function loadThemeFromPrefs(): void {
  if (_prefs === null) {
    return;
  }
  try {
    const follow: boolean = _prefs.getSync(KEY_FOLLOW_SYSTEM, true) as boolean;
    const dark: boolean = _prefs.getSync(KEY_IS_DARK, false) as boolean;
    AppStorage.setOrCreate('followSystem', follow);
    AppStorage.setOrCreate('isDarkMode', dark);
  } catch (e) {
  }
}
function saveFollowSystem(follow: boolean): void {
  if (_prefs === null) {
    return;
  }
  try {
    _prefs.putSync(KEY_FOLLOW_SYSTEM, follow);
    _prefs.flushSync();
  } catch (e) {
  }
}
function saveIsDark(dark: boolean): void {
  if (_prefs === null) {
    return;
  }
  try {
    _prefs.putSync(KEY_IS_DARK, dark);
    _prefs.flushSync();
  } catch (e) {
  }
}
export function getSystemDark(): boolean {
  return (AppStorage.get<boolean>('systemDark') ?? false) as boolean;
}
export function isDarkMode(): boolean {
  return (AppStorage.get<boolean>('isDarkMode') ?? false) as boolean;
}
export function isFollowSystem(): boolean {
  return (AppStorage.get<boolean>('followSystem') ?? true) as boolean;
}
export function toggleDarkMode(dark: boolean): void {
  AppStorage.setOrCreate('isDarkMode', dark);
  saveIsDark(dark);
  if (_appContext === null) {
    return;
  }
  try {
    _appContext.setColorMode(dark
      ? ConfigurationConstant.ColorMode.COLOR_MODE_DARK
      : ConfigurationConstant.ColorMode.COLOR_MODE_LIGHT);
  } catch (e) {
  }
}
export function setFollowSystem(follow: boolean): void {
  AppStorage.setOrCreate('followSystem', follow);
  saveFollowSystem(follow);
  // 不调 setColorMode — 避免任何副作用
  // 跟随系统时系统自动管颜色（应用未调 setColorMode 即默认跟随）
  // 手动模式时由 toggleDarkMode 调 setColorMode 强制指定
  if (follow) {
    const sysDark: boolean = getSystemDark();
    AppStorage.setOrCreate('isDarkMode', sysDark);
  }
}
